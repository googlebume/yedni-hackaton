Create a crowdfunding/project management platform for NGOs and foundations with Jira-inspired design. This is an MVP built by 2 developers in 2 days.

TECH STACK:
- Frontend: React SPA with TypeScript
- Backend: NestJS with TypeScript
- Database: Prisma ORM (PostgreSQL schema)
- Auth: bcrypt + @nestjs/jwt
- UI: Tailwind CSS + shadcn/ui components
- State: React Context API or Zustand

DATABASE SCHEMA (Prisma):
```prisma
model User {
  id            String    @id @default(uuid())
  email         String    @unique
  password      String
  name          String
  userType      UserType
  isDiiaVerified Boolean  @default(false)
  createdAt     DateTime  @default(now())
  projects      Project[]
  donations     Donation[]
}

enum UserType {
  ORGANIZATION
  DONOR
}

model Project {
  id          String   @id @default(uuid())
  title       String
  description String
  category    String[]
  goalAmount  Float
  currentAmount Float  @default(0)
  status      ProjectStatus
  creatorId   String
  creator     User     @relation(fields: [creatorId], references: [id])
  createdAt   DateTime @default(now())
  donations   Donation[]
  comments    Comment[]
}

enum ProjectStatus {
  DRAFT
  FUNDING
  IN_PROGRESS
  COMPLETED
}

model Donation {
  id        String   @id @default(uuid())
  amount    Float
  donorId   String
  donor     User     @relation(fields: [donorId], references: [id])
  projectId String
  project   Project  @relation(fields: [projectId], references: [id])
  createdAt DateTime @default(now())
}

model Comment {
  id        String   @id @default(uuid())
  content   String
  authorId  String
  projectId String
  project   Project  @relation(fields: [projectId], references: [id])
  createdAt DateTime @default(now())
}
```
